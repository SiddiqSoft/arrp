200 resource_guard(resource_guard&& src) noexcept(false)
201 try
202 : m_rsrc(std::move(src.m_rsrc))
203 , m_putback_callback(std::move(src.m_putback_callback))
204 , m_is_valid(src.m_is_valid)
205 , m_error_code(src.m_error_code) {
206 // This code is in the try block to ensure that if T's move constructor throws,
207 // we can still safely invalidate the source and prevent double-return.
208 // Its syntax is a bit unusual, but it is valid C++ and ensures exception safety.
209 // Reset to ensure that the source does not double return or preserve stale error state.
210 src.m_putback_callback = {};
211 src.m_is_valid = false;
212 src.m_error_code = pool_error::Ok;
213 }
214 catch (...) {
215 // Undo everything..
216 src.m_putback_callback = {};
217 src.m_is_valid = false;
218 src.m_error_code = pool_error::Ok;
219 // propogate back..
220 throw;
221 }
222
240 resource_guard& operator=(resource_guard&& src)
241 {
242 if (this != &src) {
243 // Return current resource to pool before overwriting it.
244 // The callback is invoked with the current validity flag so the pool
245 // can decide whether to reuse (valid) or discard (invalid) the resource.
246 if (m_putback_callback) {
247 try {
248 m_putback_callback(std::move(m_rsrc), m_is_valid);
249 }
250 catch (...) {
251 std::print(std::cerr,
252 "resource_guard move-assignment: exception while returning current resource to pool!\n");
253 }
254 // Clear the old callback and validity now that the resource has been handed back.
255 m_putback_callback = {};
256 m_is_valid = false;
257 }
258
259 // Disarm src prior to moving m_rsrc to ensure exception safety:
260 // if T's move-assignment throws, src will not double-return its resource.
261 auto src_cb = std::move(src.m_putback_callback);
262 bool src_valid = src.m_is_valid;
263 auto src_err = src.m_error_code;
264
265 src.m_putback_callback = {};
266 src.m_is_valid = false;
267 src.m_error_code = pool_error::Ok;
268
269 try {
270 m_rsrc = std::move(src.m_rsrc);
271 m_putback_callback = std::move(src_cb);
272 m_is_valid = src_valid;
273 m_error_code = src_err;
274 }
275 catch (...) {
276 m_putback_callback = {};
277 m_is_valid = false;
278 m_error_code = pool_error::Unknown;
279 throw;
280 }
281 }
282 return *this;
283 }

285 public:
297 ~resource_guard() noexcept
298 {
299 // Invoke callback if it exists, passing the resource and validity status
300 // The callback (typically resource_pool::return_to_pool) decides whether to
301 // reuse the resource (if valid) or discard it (if invalid)
302 if (m_putback_callback) {
303 try {
304 m_putback_callback(std::move(m_rsrc), m_is_valid);
305 }
306 catch (...) {
307 std::print(std::cerr, "resource_guard destructor: exception while invoking putback callback!\n");
308 }
309 m_is_valid = false;
310 m_putback_callback = {};
311 }
312 }
