248 auto make_resource_guard(Args&&... args) -> resource_guard<T>
249 {
250 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
251 return resource_guard<T> {[this](T&& src, bool isvalid) {

269 template <typename F, typename... Args>
270 auto create_from_callback(F&& f, Args&&... args) -> resource_guard<T>
271 {
272 using result_t = std::invoke_result_t<F, Args...>;
273
274 if constexpr (std::is_same_v<result_t, resource_guard<T>>) {
275 return std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
276 }
277 else if constexpr (std::is_same_v<result_t, T>) {
278 return make_resource_guard(std::invoke(std::forward<F>(f), std::forward<Args>(args)...));
279 }
280 else {
281 static_assert(std::is_same_v<result_t, resource_guard<T>> || std::is_same_v<result_t, T>,
282 "Callback must return either resource_guard<T> or T");

296 template <typename F>
297 void set_factory_callback(F&& f)
298 {
299 auto factory = std::forward<F>(f);
300 auto wrapped = [this, factory = std::move(factory)]() mutable -> resource_guard<T> {
301 return create_from_callback(factory);
302 };
303
304 // Synchronize with borrow_impl()'s reads of m_factory_callback (both the
305 // locked and unlocked read sites) so concurrent set_factory_callback()
306 // calls do not race with in-flight borrows.
307 std::scoped_lock l(m_pool_lock);
308 m_factory_callback = std::move(wrapped);
309 }
310
329 auto clear() -> pool_error
330 {
331 std::scoped_lock l(m_pool_lock);
332
333 // Drain semaphore and pool together so their counts stay in sync.
334 // The cleanup callback, if set, runs under the lock per item.
335 //
336 // @note We MUST use try_acquire() here, not the blocking acquire().
337 // A concurrent borrow_impl() may have already claimed a permit via
338 // m_pool_semaphore.try_acquire()/try_acquire_for() and be waiting on
339 // m_pool_lock (which we currently hold) before it pops its item from
340 // m_pool. If we block here waiting for a permit that will only be
341 // released once that borrower runs — and it can't run until we
342 // release this lock — the pool deadlocks. When try_acquire() fails,
343 // every remaining permit is already claimed by such an in-flight
344 // borrower, so we stop draining and let those borrowers pop their
345 // own items once they acquire the lock after we release it.
346 while (!m_pool.empty() && m_pool_semaphore.try_acquire()) {
347 auto item = std::move(m_pool.front());
348 m_pool.pop_front();
349 if (m_callback_on_resource_cleanup) {
350 try {
351 m_callback_on_resource_cleanup(item);
352 }
353 catch (std::exception& ex) {
354 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
355 }
356 catch (...) {
357 std::print(std::cerr, "{} - unknown exception while delegating to on_cleanup\n", __func__);
358 }
359 }
360 }
361
362 return pool_error::Ok;
363 }

391 [[nodiscard]] auto borrow_impl(std::chrono::nanoseconds timeout = {}, bool createIfEmptyTimeout = false)
392 -> resource_guard<T>
393 {
394 try {
395 // @note We use a unique_lock vs a scoped_lock to allow ourselves
396 // to create the resource outside the lock!
397
398 // Fast pre-check — the authoritative shutdown check is performed
399 // under the lock below after semaphore acquisition.
400 if (m_is_shutdown) return resource_guard<T> {pool_error::ShutdownInitiated};
401
402 // Wait for an available resource, then optionally create one when
403 // try_borrow_create() was requested.
404 if (timeout.count() == 0 ? m_pool_semaphore.try_acquire() : m_pool_semaphore.try_acquire_for(timeout)) {
405 // We likely have a resource available.. grab a lock.
406 std::unique_lock l(m_pool_lock);
407
408 // Authoritative shutdown check under the lock: the destructor may
409 // have run between the pre-semaphore check and taking this lock.
410 if (m_is_shutdown) {
411 m_pool_semaphore.release();
412 return resource_guard<T> {pool_error::ShutdownInitiated};
413 }
414
415 // We have a resource available.. just to be sure, we'll check the pool size inside the lock..
416 if (!m_pool.empty()) {
417 // Move the resource out of the pool while holding the lock.
418 auto borrowed = make_resource_guard(std::move(m_pool.front()));
419 // Remove from the deque.. only one client may have exclusive use..
420 m_pool.pop_front();
421
422 // Increment checkout and borrow counters under the lock to prevent
423 // state accounting underflow / race conditions with concurrent return_to_pool()
424 m_resources_checkedout++;
425 m_counter_borrows++;
426
427 l.unlock();
428
429 return borrowed;
430 }
431 else if (createIfEmptyTimeout && m_factory_callback) {
432#if defined(arrp_DEBUG_TRACE)
433 std::println(std::cerr, "{} - Empty pool; asked to create new if empty..", __func__);
434#endif
435
436 // Release the lock before invoking arbitrary user code to
437 // prevent deadlock if the factory calls back into the pool.
438 auto cb = m_factory_callback;
439 l.unlock();
440 auto ondemand = create_from_callback(cb);
441 m_counter_ondemand_adds++;
442 m_resources_checkedout++;
443 m_counter_borrows++;
444 return ondemand;
445 }
446 else {
447 // Permit was acquired but no resource exists; release it so
448 // the semaphore count stays in sync with the pool size.
449 m_pool_semaphore.release();
450 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
451 }
452 }
453 else if (createIfEmptyTimeout) {
454 // Take a copy of the factory under the lock so the read is
455 // synchronized with set_factory_callback(), then invoke it
456 // without holding m_pool_lock to avoid deadlock if the factory
457 // calls back into the pool.
458 std::function<resource_guard<T>()> cb;
459 {
460 std::scoped_lock l(m_pool_lock);
461 cb = m_factory_callback;
462 }
463 if (!cb) {
464 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
465 }
466
467#if defined(arrp_DEBUG_TRACE)
468 std::println(std::cerr, "{} - We exhausted timeout; asked to create new if empty..", __func__);
469#endif
470
471 auto ondemand = create_from_callback(cb);
472 m_counter_ondemand_adds++;
473 m_resources_checkedout++;
474 m_counter_borrows++;
475 return ondemand;
476 }
477 else if (timeout.count() > 0) {
478 return resource_guard<T> {siddiqsoft::arrp::pool_error::Timeout};
479 }
480 else {
481 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
482 }
483 } // scope end
484 catch (std::exception& ex) {
485 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
486 }
487 catch (...) {
488 std::println(std::cerr, "UNKNOWN Error in borrow");
489 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
490 }
491
492 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
493 }

529 template <typename... Args>
530 auto seed(Args&&... args) -> pool_error
531 {
532 {
533 std::scoped_lock l(m_pool_lock);
534
535 // Check inside the lock..
536 if (m_is_shutdown) return pool_error::ShutdownInitiated;
537
538 m_pool.emplace_back(std::forward<Args>(args)...);
539 m_counter_seeds++;
540
541 // Update peak pool size for statistics
542 auto current_size = m_pool.size();
543 if (current_size > m_peak_poolsize.load()) {
544 m_peak_poolsize = current_size;
545 }
546
547 // Signal semaphore while holding the lock to maintain atomic visibility
548 m_pool_semaphore.release();
549 }
550
551 return pool_error::Ok;
552 }

587 protected:
598 void return_to_pool(T&& item, bool isvalid)
599 {
600 std::unique_lock l(m_pool_lock);
601
602 // Always decrement the checkout counter — the resource is leaving our
603 // custody regardless of whether the pool is still alive.
604 m_resources_checkedout--;
605
606 // Check inside the lock..
607 if (m_is_shutdown) return;
608
609 if (isvalid) {
610 m_pool.push_back(std::move(item));
611 m_counter_returns++;
612
613 // Update peak pool size for statistics
614 auto current_size = m_pool.size();
615 if (current_size > m_peak_poolsize.load()) {
616 m_peak_poolsize = current_size;
617 }
618
619 // Signal semaphore while holding the lock to maintain atomic visibility
620 m_pool_semaphore.release();
621 }
622 else {
623 m_counter_abandons++;
624 }
625 }
