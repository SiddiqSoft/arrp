280 template <typename F, typename... Args>
281 auto create_from_callback(F&& f, Args&&... args) -> resource_guard<T>
282 {
283 using result_t = std::invoke_result_t<F, Args...>;
284
285 if constexpr (std::is_same_v<result_t, resource_guard<T>>) {
286 return std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
287 }
288 else if constexpr (std::is_same_v<result_t, T>) {
289 return make_resource_guard(std::invoke(std::forward<F>(f), std::forward<Args>(args)...));
290 }
291 else {
292 static_assert(std::is_same_v<result_t, resource_guard<T>> || std::is_same_v<result_t, T>,
293 "Callback must return either resource_guard<T> or T");

307 template <typename F>
308 void set_factory_callback(F&& f)
309 {
310 auto factory = std::forward<F>(f);
311 auto wrapped = [this, factory = std::move(factory)]() mutable -> resource_guard<T> {
312 return create_from_callback(factory);
313 };
314
315 // Synchronize with borrow_impl()'s reads of m_factory_callback (both the
316 // locked and unlocked read sites) so concurrent set_factory_callback()
317 // calls do not race with in-flight borrows.
318 std::scoped_lock l(m_pool_lock);
319 m_factory_callback = std::move(wrapped);
320 }
321
337 auto clear() -> pool_error
338 {
339 std::scoped_lock l(m_pool_lock);
340
341 // Drain semaphore and pool together so their counts stay in sync.
342 // The cleanup callback, if set, runs under the lock per item.
343 //
344 // @note We MUST use try_acquire() here, not the blocking acquire().
345 // A concurrent borrow_impl() may have already claimed a permit via
346 // m_pool_semaphore.try_acquire()/try_acquire_for() and be waiting on
347 // m_pool_lock (which we currently hold) before it pops its item from
348 // m_pool. If we block here waiting for a permit that will only be
349 // released once that borrower runs — and it can't run until we
350 // release this lock — the pool deadlocks. When try_acquire() fails,
351 // every remaining permit is already claimed by such an in-flight
352 // borrower, so we stop draining and let those borrowers pop their
353 // own items once they acquire the lock after we release it.
354 while (!m_pool.empty() && m_pool_semaphore.try_acquire()) {
355 auto item = std::move(m_pool.front());
356 m_pool.pop_front();
357 if (m_callback_on_resource_cleanup) {
358 try {
359 m_callback_on_resource_cleanup(item);
360 }
361 catch (std::exception& ex) {
362 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
363 }
364 catch (...) {
365 std::print(std::cerr, "{} - unknown exception while delegating to on_cleanup\n", __func__);
366 }
367 }
368 }
369
370 return pool_error::Ok;
371 }

406 [[nodiscard]] auto borrow_impl(std::chrono::nanoseconds timeout = {}, bool createIfEmptyTimeout = false)
407 -> resource_guard<T>
408 {
409 try {
410 // @note We use a unique_lock vs a scoped_lock to allow ourselves
411 // to create the resource outside the lock!
412
413 // Fast pre-check — the authoritative shutdown check is performed
414 // under the lock below after semaphore acquisition.
415 if (m_is_shutdown) return resource_guard<T> {pool_error::ShutdownInitiated};
416
417 // Wait for an available resource, then optionally create one when
418 // try_borrow_create() was requested.
419 if (timeout.count() == 0 ? m_pool_semaphore.try_acquire() : m_pool_semaphore.try_acquire_for(timeout)) {
420 // We likely have a resource available.. grab a lock.
421 std::unique_lock l(m_pool_lock);
422
423 // Authoritative shutdown check under the lock: the destructor may
424 // have run between the pre-semaphore check and taking this lock.
425 if (m_is_shutdown) {
426 m_pool_semaphore.release();
427 return resource_guard<T> {pool_error::ShutdownInitiated};
428 }
429
430 // We have a resource available.. just to be sure, we'll check the pool size inside the lock..
431 if (!m_pool.empty()) {
432 // Move the resource out of the pool while holding the lock.
433 auto borrowed = make_resource_guard(std::move(m_pool.front()));
434 // Remove from the deque.. only one client may have exclusive use..
435 m_pool.pop_front();
436
437 // Increment checkout and borrow counters under the lock to prevent
438 // state accounting underflow / race conditions with concurrent return_to_pool()
439 m_resources_checkedout++;
440 m_counter_borrows++;
441
442 l.unlock();
443
444 return borrowed;
445 }
446 else if (createIfEmptyTimeout && m_factory_callback) {
447#if defined(arrp_DEBUG_TRACE)
448 std::println(std::cerr, "{} - Empty pool; asked to create new if empty..", __func__);
449#endif
450
451 // Release the lock before invoking arbitrary user code to
452 // prevent deadlock if the factory calls back into the pool.
453 auto cb = m_factory_callback;
454 l.unlock();
455 auto ondemand = create_from_callback(cb);
456 m_counter_ondemand_adds++;
457 m_resources_checkedout++;
458 m_counter_borrows++;
459 return ondemand;
460 }
461 else {
462 // Permit was acquired but no resource exists; release it so
463 // the semaphore count stays in sync with the pool size.
464 m_pool_semaphore.release();
465 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
466 }
467 }
468 else if (createIfEmptyTimeout) {
469 // Take a copy of the factory under the lock so the read is
470 // synchronized with set_factory_callback(), then invoke it
471 // without holding m_pool_lock to avoid deadlock if the factory
472 // calls back into the pool.
473 std::function<resource_guard<T>()> cb;
474 {
475 std::scoped_lock l(m_pool_lock);
476 cb = m_factory_callback;
477 }
478 if (!cb) {
479 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
480 }
481
482#if defined(arrp_DEBUG_TRACE)
483 std::println(std::cerr, "{} - We exhausted timeout; asked to create new if empty..", __func__);
484#endif
485
486 auto ondemand = create_from_callback(cb);
487 m_counter_ondemand_adds++;
488 m_resources_checkedout++;
489 m_counter_borrows++;
490 return ondemand;
491 }
492 else if (timeout.count() > 0) {
493 return resource_guard<T> {siddiqsoft::arrp::pool_error::Timeout};
494 }
495 else {
496 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
497 }
498 } // scope end
499 catch (std::exception& ex) {
500 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
501 }
502 catch (...) {
503 std::println(std::cerr, "UNKNOWN Error in borrow");
504 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
505 }
506
507 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
508 }

535 template <typename... Args>
536 auto seed(Args&&... args) -> pool_error
537 {
538 {
539 std::scoped_lock l(m_pool_lock);
540
541 // Check inside the lock..
542 if (m_is_shutdown) return pool_error::ShutdownInitiated;
543
544 m_pool.emplace_back(std::forward<Args>(args)...);
545 m_counter_seeds++;
546
547 // Update peak pool size for statistics
548 auto current_size = m_pool.size();
549 if (current_size > m_peak_poolsize.load()) {
550 m_peak_poolsize = current_size;
551 }
552
553 // Signal semaphore while holding the lock to maintain atomic visibility
554 m_pool_semaphore.release();
555 }
556
557 return pool_error::Ok;
558 }

604 void return_to_pool(T&& item, bool isvalid)
605 {
606 std::unique_lock l(m_pool_lock);
607
608 // Always decrement the checkout counter — the resource is leaving our
609 // custody regardless of whether the pool is still alive.
610 m_resources_checkedout--;
611
612 // Check inside the lock..
613 if (m_is_shutdown) return;
614
615 if (isvalid) {
616 m_pool.push_back(std::move(item));
617 m_counter_returns++;
618
619 // Update peak pool size for statistics
620 auto current_size = m_pool.size();
621 if (current_size > m_peak_poolsize.load()) {
622 m_peak_poolsize = current_size;
623 }
624
625 // Signal semaphore while holding the lock to maintain atomic visibility
626 m_pool_semaphore.release();
627 }
628 else {
629 m_counter_abandons++;
630 }
631 }
