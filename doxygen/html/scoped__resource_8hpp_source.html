240 scoped_resource& operator=(scoped_resource&& src)
241 {
242 if (this != &src) {
243 // Return current resource to pool before overwriting it.
244 // The callback is invoked with the current validity flag so the pool
245 // can decide whether to reuse (valid) or discard (invalid) the resource.
246 if (m_putback_callback) {
247 try {
248 m_putback_callback(std::move(m_rsrc), m_is_valid);
249 }
250 catch (...) {
251 std::print(std::cerr,
252 "scoped_resource move-assignment: exception while returning current resource to pool!\n");
253 }
254 // Clear our own callback and validity now that the resource has been
255 // handed back; this prevents the destructor from double-returning.
256 m_putback_callback = {};
257 m_is_valid = false;
258 }
259
260 // Take ownership of src's resource and callback.
261 m_rsrc = std::move(src.m_rsrc);
262 m_putback_callback = std::move(src.m_putback_callback);
263 m_is_valid = src.m_is_valid;
264
265 // Disarm src so its destructor does nothing.
266 src.m_putback_callback = {};
267 src.m_is_valid = false;
268 }
269 return *this;
270 }
271
272 public:
284 ~scoped_resource() noexcept
285 {
286 // Invoke callback if it exists, passing the resource and validity status
287 // The callback (typically resource_pool::return_to_pool) decides whether to
288 // reuse the resource (if valid) or discard it (if invalid)
289 if (m_putback_callback) {
290 try {
291 m_putback_callback(std::move(m_rsrc), m_is_valid);
292 }
293 catch (...) {
294 std::print(std::cerr, "scoped_resource destructor: exception while invoking putback callback!\n");
295 }
296 m_is_valid = false;
297 m_putback_callback = {};
298 }
