284 scoped_resource& operator=(scoped_resource&& src) noexcept
285 {
286 if (this != &src) {
287 // Return current resource if valid
288 if (m_is_valid && m_putback_callback) {
289 m_putback_callback(std::move(m_rsrc));
290 }
291
292 // Move from src
293 m_rsrc = std::move(src.m_rsrc);
294 m_putback_callback = std::move(src.m_putback_callback);
295 m_abandon_callback = std::move(src.m_abandon_callback);
296 m_is_valid = src.m_is_valid;
297
298 // Reset src
299 src.m_putback_callback = {};
300 src.m_abandon_callback = {};
301 src.m_is_valid = false;
302 }
303 return *this;
304 }

397 ~scoped_resource()
398 {
399 // Only return resource if it's valid and callback exists
400 // This prevents returning uninitialized or moved-out resources to the pool
401 if (m_is_valid && m_putback_callback) {
402 m_putback_callback(std::move(m_rsrc));
403 m_is_valid = false;
404 m_putback_callback = {};
405 }
406 else if (m_abandon_callback) {
407 // Inform the pool that resource was invalidated
408 m_abandon_callback(std::move(m_rsrc));
409 m_is_valid = false;
410 m_putback_callback = {};
411 m_abandon_callback = {};
412 }
