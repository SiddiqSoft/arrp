229 scoped_resource& operator=(scoped_resource&& src)
230 {
231 if (this != &src) {
232 // Return current resource to pool before overwriting it.
233 // The callback is invoked with the current validity flag so the pool
234 // can decide whether to reuse (valid) or discard (invalid) the resource.
235 if (m_putback_callback) {
236 try {
237 m_putback_callback(std::move(m_rsrc), m_is_valid);
238 }
239 catch (...) {
240 std::print(std::cerr,
241 "scoped_resource move-assignment: exception while returning current resource to pool!\n");
242 }
243 // Clear our own callback and validity now that the resource has been
244 // handed back; this prevents the destructor from double-returning.
245 m_putback_callback = {};
246 m_is_valid = false;
247 }
248
249 // Take ownership of src's resource and callback.
250 m_rsrc = std::move(src.m_rsrc);
251 m_putback_callback = std::move(src.m_putback_callback);
252 m_is_valid = src.m_is_valid;
253
254 // Disarm src so its destructor does nothing.
255 src.m_putback_callback = {};
256 src.m_is_valid = false;
257 }
258 return *this;
259 }

311 ~scoped_resource() noexcept
312 {
313 // Invoke callback if it exists, passing the resource and validity status
314 // The callback (typically resource_pool::return_to_pool) decides whether to
315 // reuse the resource (if valid) or discard it (if invalid)
316 if (m_putback_callback) {
317 try {
318 m_putback_callback(std::move(m_rsrc), m_is_valid);
319 }
320 catch (...) {
321 std::print(std::cerr, "scoped_resource destructor: exception while invoking putback callback!\n");
322 }
323 m_is_valid = false;
324 m_putback_callback = {};
325 }
