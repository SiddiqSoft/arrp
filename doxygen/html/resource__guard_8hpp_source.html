248 resource_guard& operator=(resource_guard&& src)
249 {
250 if (this != &src) {
251 // Return current resource to pool before overwriting it.
252 // The callback is invoked with the current validity flag so the pool
253 // can decide whether to reuse (valid) or discard (invalid) the resource.
254 if (m_putback_callback) {
255 try {
256 m_putback_callback(std::move(m_rsrc), m_is_valid);
257 }
258 catch (...) {
259 std::print(std::cerr,
260 "resource_guard move-assignment: exception while returning current resource to pool!\n");
261 }
262 // Clear the old callback and validity now that the resource has been handed back.
263 m_putback_callback = {};
264 m_is_valid = false;
265 }
266
267 // Take ownership of src's resource and callback.
268 m_rsrc = std::move(src.m_rsrc);
269 m_putback_callback = std::move(src.m_putback_callback);
270 m_is_valid = src.m_is_valid;
271 m_error_code = src.m_error_code;
272
273 // Disarm src so its destructor does nothing and stale error state is cleared.
274 src.m_putback_callback = {};
275 src.m_is_valid = false;
276 src.m_error_code = pool_error::Ok;
277 }
278 return *this;
279 }

293 ~resource_guard() noexcept
294 {
295 // Invoke callback if it exists, passing the resource and validity status
296 // The callback (typically resource_pool::return_to_pool) decides whether to
297 // reuse the resource (if valid) or discard it (if invalid)
298 if (m_putback_callback) {
299 try {
300 m_putback_callback(std::move(m_rsrc), m_is_valid);
301 }
302 catch (...) {
303 std::print(std::cerr, "resource_guard destructor: exception while invoking putback callback!\n");
304 }
305 m_is_valid = false;

307 }
