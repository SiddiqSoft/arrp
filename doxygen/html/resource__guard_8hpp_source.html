219 resource_guard(resource_guard&& src) noexcept(std::is_nothrow_move_constructible_v<T> &&
220 std::is_nothrow_move_constructible_v<PutbackCallbackFunc>)
221 try
222 : m_rsrc(std::move(src.m_rsrc))
223 , m_putback_callback(std::move(src.m_putback_callback))
224 , m_is_valid(src.m_is_valid)
225 , m_error_code(src.m_error_code) {
226 // Reset to ensure that the source does not double return or preserve stale error state.
227 src.m_putback_callback = {};
228 src.m_is_valid = false;
229 src.m_error_code = pool_error::Ok;
230 }
231 catch (...) {
232 // Undo everything..
233 src.m_putback_callback = {};
234 src.m_is_valid = false;
235 src.m_error_code = pool_error::Ok;
236 // propogate back..
237 throw;
238 }

257 resource_guard& operator=(resource_guard&& src)
258 {
259 if (this != &src) {
260 // Return current resource to pool before overwriting it.
261 // The callback is invoked with the current validity flag so the pool
262 // can decide whether to reuse (valid) or discard (invalid) the resource.
263 if (m_putback_callback) {
264 try {
265 m_putback_callback(std::move(m_rsrc), m_is_valid);
266 }
267 catch (...) {
268 std::print(std::cerr,
269 "resource_guard move-assignment: exception while returning current resource to pool!\n");
270 }
271 // Clear the old callback and validity now that the resource has been handed back.
272 m_putback_callback = {};
273 m_is_valid = false;
274 }
275
276 // Disarm src prior to moving m_rsrc to ensure exception safety:
277 // if T's move-assignment throws, src will not double-return its resource.
278 auto src_cb = std::move(src.m_putback_callback);
279 bool src_valid = src.m_is_valid;
280 auto src_err = src.m_error_code;
281
282 src.m_putback_callback = {};
283 src.m_is_valid = false;
284 src.m_error_code = pool_error::Ok;
285
286 try {
287 m_rsrc = std::move(src.m_rsrc);
288 m_putback_callback = std::move(src_cb);
289 m_is_valid = src_valid;
290 m_error_code = src_err;
291 }
292 catch (...) {
293 m_putback_callback = {};
294 m_is_valid = false;
295 m_error_code = pool_error::Unknown;
296 throw;
297 }
298 }
299 return *this;
300 }

314 ~resource_guard() noexcept
315 {
316 // Invoke callback if it exists, passing the resource and validity status
317 // The callback (typically resource_pool::return_to_pool) decides whether to
318 // reuse the resource (if valid) or discard it (if invalid)
319 if (m_putback_callback) {
320 try {
321 m_putback_callback(std::move(m_rsrc), m_is_valid);
322 }
323 catch (...) {
324 std::print(std::cerr, "resource_guard destructor: exception while invoking putback callback!\n");
325 }
326 m_is_valid = false;
327 m_putback_callback = {};
328 }
