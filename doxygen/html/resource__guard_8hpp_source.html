217 resource_guard(resource_guard&& src) noexcept(std::is_nothrow_move_constructible_v<T> &&
218 std::is_nothrow_move_constructible_v<PutbackCallbackFunc>)
219 : m_rsrc(std::move(src.m_rsrc))
220 , m_putback_callback(std::move(src.m_putback_callback))
221 , m_is_valid(src.m_is_valid)
222 , m_error_code(src.m_error_code)
223 {
224 // Reset to ensure that the source does not double return or preserve stale error state.
225 src.m_putback_callback = {};
226 src.m_is_valid = false;
227 src.m_error_code = pool_error::Ok;

247 resource_guard& operator=(resource_guard&& src)
248 {
249 if (this != &src) {
250 // Return current resource to pool before overwriting it.
251 // The callback is invoked with the current validity flag so the pool
252 // can decide whether to reuse (valid) or discard (invalid) the resource.
253 if (m_putback_callback) {
254 try {
255 m_putback_callback(std::move(m_rsrc), m_is_valid);
256 }
257 catch (...) {
258 std::print(std::cerr,
259 "resource_guard move-assignment: exception while returning current resource to pool!\n");
260 }
261 // Clear the old callback and validity now that the resource has been handed back.
262 m_putback_callback = {};
263 m_is_valid = false;
264 }
265
266 // Take ownership of src's resource and callback.
267 m_rsrc = std::move(src.m_rsrc);
268 m_putback_callback = std::move(src.m_putback_callback);
269 m_is_valid = src.m_is_valid;
270 m_error_code = src.m_error_code;
271
272 // Disarm src so its destructor does nothing and stale error state is cleared.
273 src.m_putback_callback = {};
274 src.m_is_valid = false;
275 src.m_error_code = pool_error::Ok;
276 }
277 return *this;
278 }
279
280 public:
292 ~resource_guard() noexcept
293 {
294 // Invoke callback if it exists, passing the resource and validity status
295 // The callback (typically resource_pool::return_to_pool) decides whether to
296 // reuse the resource (if valid) or discard it (if invalid)
297 if (m_putback_callback) {
298 try {
299 m_putback_callback(std::move(m_rsrc), m_is_valid);
300 }
301 catch (...) {
302 std::print(std::cerr, "resource_guard destructor: exception while invoking putback callback!\n");
303 }
304 m_is_valid = false;
305 m_putback_callback = {};
306 }
307 }

368 resource_guard& operator=(T&& src)
369 {
370 // Replace the held value in place. The guard retains ownership via
371 // m_putback_callback and will return the new resource on destruction.
372 // Invoking the callback here would cause a double-return: once now
373 // and again when the destructor fires.
