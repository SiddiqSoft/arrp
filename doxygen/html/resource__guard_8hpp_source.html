215 resource_guard(resource_guard&& src) noexcept(
216 std::is_nothrow_move_constructible_v<T> && std::is_nothrow_move_constructible_v<PutbackCallbackFunc>)
217 : m_rsrc(std::move(src.m_rsrc))
218 , m_putback_callback(std::move(src.m_putback_callback))
219 , m_is_valid(src.m_is_valid)
220 , m_error_code(src.m_error_code)
221 {
222 // Reset to ensure that the source does not double return or preserve stale error state.
223 src.m_putback_callback = {};
224 src.m_is_valid = false;
225 src.m_error_code = pool_error::Ok;
226 }

245 resource_guard& operator=(resource_guard&& src)
246 {
247 if (this != &src) {
248 // Return current resource to pool before overwriting it.
249 // The callback is invoked with the current validity flag so the pool
250 // can decide whether to reuse (valid) or discard (invalid) the resource.
251 if (m_putback_callback) {
252 try {
253 m_putback_callback(std::move(m_rsrc), m_is_valid);
254 }
255 catch (...) {
256 std::print(std::cerr,
257 "resource_guard move-assignment: exception while returning current resource to pool!\n");
258 }
259 // Clear the old callback and validity now that the resource has been handed back.
260 m_putback_callback = {};
261 m_is_valid = false;
262 }
263
264 // Take ownership of src's resource and callback.
265 m_rsrc = std::move(src.m_rsrc);
266 m_putback_callback = std::move(src.m_putback_callback);
267 m_is_valid = src.m_is_valid;
268 m_error_code = src.m_error_code;
269
270 // Disarm src so its destructor does nothing and stale error state is cleared.
271 src.m_putback_callback = {};
272 src.m_is_valid = false;
273 src.m_error_code = pool_error::Ok;
274 }
275 return *this;
276 }

290 ~resource_guard() noexcept
291 {
292 // Invoke callback if it exists, passing the resource and validity status
293 // The callback (typically resource_pool::return_to_pool) decides whether to
294 // reuse the resource (if valid) or discard it (if invalid)
295 if (m_putback_callback) {
296 try {
297 m_putback_callback(std::move(m_rsrc), m_is_valid);
298 }
299 catch (...) {
300 std::print(std::cerr, "resource_guard destructor: exception while invoking putback callback!\n");
301 }
302 m_is_valid = false;
303 m_putback_callback = {};
304 }
