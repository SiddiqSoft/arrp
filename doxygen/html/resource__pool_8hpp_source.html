196 inline auto loan_size() const

277 auto make_resource_guard(Args&&... args) -> SRT
278 {
279 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
280 return SRT {[this](T&& src, bool isvalid) {

299 auto create_from_callback(F&& f, Args&&... args) -> SRT
300 {
301 using result_t = std::invoke_result_t<F, Args...>;
302
303 if constexpr (std::is_same_v<result_t, SRT>) {
304 return std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
305 }
306 else if constexpr (std::is_same_v<result_t, T>) {
307 return make_resource_guard(std::invoke(std::forward<F>(f), std::forward<Args>(args)...));

319 template <typename F>
320 void set_factory_callback(F&& f)
321 {
322 auto factory = std::forward<F>(f);
323 auto wrapped = [this, factory = std::move(factory)]() mutable -> SRT {
324 return create_from_callback(factory);
325 };
326
327 // Synchronize with borrow_impl()'s reads of m_factory_callback (both the
328 // locked and unlocked read sites) so concurrent set_factory_callback()
329 // calls do not race with in-flight borrows.
330 std::scoped_lock l(m_pool_lock);
331 m_factory_callback = std::move(wrapped);
332 }
333
343 auto clear() -> pool_error
344 {
345 std::scoped_lock l(m_pool_lock);
346
347 try {
348 // Drain semaphore and pool together so their counts stay in sync.
349 // The cleanup callback, if set, runs under the lock per item.
350 //
351 // @note We MUST use try_acquire() here, not the blocking acquire().
352 // A concurrent borrow_impl() may have already claimed a permit via
353 // m_pool_semaphore.try_acquire()/try_acquire_for() and be waiting on
354 // m_pool_lock (which we currently hold) before it pops its item from
355 // m_pool. If we block here waiting for a permit that will only be
356 // released once that borrower runs — and it can't run until we
357 // release this lock — the pool deadlocks. When try_acquire() fails,
358 // every remaining permit is already claimed by such an in-flight
359 // borrower, so we stop draining and let those borrowers pop their
360 // own items once they acquire the lock after we release it.
361 while (!m_pool.empty() && m_pool_semaphore.try_acquire()) {
362 auto item = std::move(m_pool.front());
363 m_pool.pop_front();
364 if (m_callback_on_resource_cleanup) {
365 // delegate to the cleanup.
366 // the delegate must not invoke any pool member to avoid deadlocks.
367 m_callback_on_resource_cleanup(item);
368 }
369 }
370 }
371 catch (std::exception& ex) {
372 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
373 }
374
375 return pool_error::Ok;
376 }

411 [[nodiscard]] auto borrow_impl(std::chrono::nanoseconds timeout = {}, bool createIfEmptyTimeout = false) -> SRT
412 {
413 try {
414 // @note We use a unique_lock vs a scoped_lock to allow ourselves
415 // to create the resource outside the lock!
416
417 // Fast pre-check — the authoritative shutdown check is performed
418 // under the lock below after semaphore acquisition.
419 if (m_is_shutdown) return SRT {pool_error::ShutdownInitiated};
420
421 // Wait for an available resource, then optionally create one when
422 // try_borrow_create() was requested.
423 if (timeout.count() == 0 ? m_pool_semaphore.try_acquire() : m_pool_semaphore.try_acquire_for(timeout)) {
424 // We likely have a resource available.. grab a lock.
425 std::unique_lock l(m_pool_lock);
426
427 // Authoritative shutdown check under the lock: the destructor may
428 // have run between the pre-semaphore check and taking this lock.
429 if (m_is_shutdown) {
430 m_pool_semaphore.release();
431 return SRT {pool_error::ShutdownInitiated};
432 }
433
434 // We have a resource available.. just to be sure, we'll check the pool size inside the lock..
435 if (!m_pool.empty()) {
436 // Move the resource out of the pool while holding the lock.
437 auto borrowed = make_resource_guard(std::move(m_pool.front()));
438 // Remove from the deque.. only one client may have exclusive use..
439 m_pool.pop_front();
440
441 // Release the lock before constructing the scoped wrapper
442 // and updating the borrow counters.
443 l.unlock();
444
445 m_resources_checkedout++;
446 m_counter_borrows++;
447 return borrowed;
448 }
449 else if (createIfEmptyTimeout && m_factory_callback) {
450 std::println(std::cerr, "{} - Empty pool; asked to create new if empty..", __func__);
451 // Release the lock before invoking arbitrary user code to
452 // prevent deadlock if the factory calls back into the pool.
453 auto cb = m_factory_callback;
454 l.unlock();
455 auto ondemand = create_from_callback(cb);
456 m_counter_ondemand_adds++;
457 m_resources_checkedout++;
458 m_counter_borrows++;
459 return ondemand;
460 }
461 else {
462 // Permit was acquired but no resource exists; release it so
463 // the semaphore count stays in sync with the pool size.
464 m_pool_semaphore.release();
465 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
466 }
467 }
468 else if (createIfEmptyTimeout) {
469 // Take a copy of the factory under the lock so the read is
470 // synchronized with set_factory_callback(), then invoke it
471 // without holding m_pool_lock to avoid deadlock if the factory
472 // calls back into the pool.
473 std::function<SRT()> cb;
474 {
475 std::scoped_lock l(m_pool_lock);
476 cb = m_factory_callback;
477 }
478 if (!cb) {
479 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
480 }
481 std::println(std::cerr, "{} - We exhausted timeout; asked to create new if empty..", __func__);
482 auto ondemand = create_from_callback(cb);
483 m_counter_ondemand_adds++;
484 m_resources_checkedout++;
485 m_counter_borrows++;
486 return ondemand;
487 }
488 else if (timeout.count() > 0) {
489 return SRT {siddiqsoft::arrp::pool_error::Timeout};
490 }
491 else {
492 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
493 }
494 } // scope end
495 catch (std::exception& ex) {
496 return SRT {siddiqsoft::arrp::pool_error::Unknown};
497 }
498 catch (...) {
499 std::println(std::cerr, "UNKNOWN Error in borrow");
500 return SRT {siddiqsoft::arrp::pool_error::Unknown};
501 }
502
503 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
504 }

528 template <typename... Args>
529 auto seed(Args&&... args) -> pool_error
530 {
531 {
532 std::scoped_lock l(m_pool_lock);
533
534 // Check inside the lock..
535 if (m_is_shutdown) return pool_error::ShutdownInitiated;
536
537 m_pool.emplace_back(std::forward<Args>(args)...);
538 m_counter_seeds++;
539
540 // Update peak pool size for statistics
541 auto current_size = m_pool.size();
542 if (current_size > m_peak_poolsize.load()) {
543 m_peak_poolsize = current_size;
544 }
545 }
546
547 // Signal the semaphore outside the lock to avoid potential deadlocks.
548 m_pool_semaphore.release(); // resource is available, increment semaphore
549
550 return pool_error::Ok;
551 }

597 void return_to_pool(T&& item, bool isvalid)
598 {
599 std::unique_lock l(m_pool_lock);
600
601 // Always decrement the checkout counter — the resource is leaving our
602 // custody regardless of whether the pool is still alive.
603 m_resources_checkedout--;
604
605 // Check inside the lock..
606 if (m_is_shutdown) return;
607
608 if (isvalid) {
609 m_pool.push_back(std::move(item));
610 m_counter_returns++;
611
612 // Update peak pool size for statistics
613 auto current_size = m_pool.size();
614 if (current_size > m_peak_poolsize.load()) {
615 m_peak_poolsize = current_size;
616 }
617
618 l.unlock();
619 // Signal the semaphore outside the lock to avoid potential deadlocks.
620 if (isvalid) m_pool_semaphore.release(); // resource is available, increment semaphore
621 }
622 else {
623 m_counter_abandons++;
624 }
625 }
