191 inline int64_t deficit_size()

200 inline auto loan_size()
201 {
202 auto loans = m_counter_borrows.load(); // total number of borrows (current counter)
203 loans -= m_counter_returns.load(); // total number of returns (current counter)
204 loans -= m_counter_abandons.load(); // adjust for any abandons
205 return loans;
206 }

265 resource_pool(uint8_t init_capacity = resource_pool_limits::DefaultCapacity,
266 auto_add_policy add_policy = auto_add_policy::NoGrow)
267 {
268#if defined(DEBUG)
269 std::print(std::cerr,
270 "{}(x,b) - Invoked; init_capacity:{} with new add_policy: {}\n",
271 __func__,
272 init_capacity,
273 add_policy);
274#endif
275 set_capacity(init_capacity);
276
277 if (add_policy == auto_add_policy::NoGrow) {
278 m_callback_to_add_new_raw_resource_to_pool = CallbackDoNotAutoAddResource;
279 }
280 else if (add_policy == auto_add_policy::AutoGrow) {
281 // This method is declared here as lambda to capture the this pointer
282 // whereas if we attempted to declared it earlier as a static inline then the
283 // this pointer would not be captured.
284 m_callback_to_add_new_raw_resource_to_pool = [this](resource_pool& pool) -> std::expected<SRT, pool_error> {
285 // Create a SRT element and wire up the auto-return callback to return
286 // the resource back to this object.
287 return SRT {[this](T&& src, bool isvalid) {
288 // this callback puts the resource back..
289 return this->return_to_pool(std::forward<T&&>(src), isvalid);
290 },
291 T {}};
292 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
293 };
294 }

344 auto clear() -> std::expected<void, pool_error>
345 {
346 std::scoped_lock l(m_pool_lock);
347
348#if defined(DEBUG)
349 std::print(std::cerr, "{} - invoked; size:{} is shutdown? {}\n", __func__, m_pool.size(), m_is_shutdown.load());
350#endif
351
352 try {
353 if (m_callback_on_resource_cleanup && !m_pool.empty()) {
354 while (!m_pool.empty()) {
355 RunOnEnd roe([&] { m_pool.pop_front(); });
356 // delegate to the cleanup.
357 // the delegate must not invoke any pool member to avoid deadlocks.
358 m_callback_on_resource_cleanup(std::move(m_pool.front()));
359 }
360 }
361 }
362 catch (std::exception& ex) {
363 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
364 }
365
366 m_pool.clear();
367
368 return {};
369 }

409 [[nodiscard]] auto borrow_from_pool() -> std::expected<SRT, pool_error>
410 {
411 try {
412 // @note We use a unique_lock vs a scoped_lock to allow ourselves
413 // to create the resource outside the lock!
414 std::unique_lock l(m_pool_lock);
415
416 // Now that we're inside the lock, we should check again to ensure that
417 // the shutdown is not in progress..
418 if (m_is_shutdown) return std::unexpected(pool_error::ShutdownInitiated);
419
420 if (!m_pool.empty()) {
421 // The pool is non-empty; return from the pool
422 // Return first element from the pool and pop it on scope end
423 RunOnEnd pop_guard([&]() {
424 m_pool.pop_front();
425 m_resources_checkedout++;
426 m_counter_borrows++;
427 });
428
429 // Make a wrapper..
430 // Create a SRT element and wire up the auto-return callback to return
431 // the resource back to this object.
432 return SRT {[this](T&& src, bool isvalid) {
433 // this callback puts the resource back..
434 return this->return_to_pool(std::forward<T&&>(src), isvalid);
435 },
436 std::move(m_pool.front())};
437 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
438 // The pop_front() happens within this scope and within the lock!
439 }
440 else if (is_pool_starving() && m_callback_to_add_new_raw_resource_to_pool) {
441 // We have no more items in the pool (we're starting up or everything is
442 // checked out) but we have not reached the limit. The limit is number
443 // of m_resources_checkedout + pool.size() < m_capacity We are
444 // under-capacity.. so we can return to the caller a new item..
445 // We should unlock the resource and ..
446 l.unlock();
447
448 // Update the attempted delegated calls to add new raw resource to pool.
449 // ..delegate the new resource acquisition outside the lock.
450 return m_callback_to_add_new_raw_resource_to_pool(*this).and_then(
451 [&](auto item) -> std::expected<SRT, pool_error> {
452 m_resources_checkedout++;
453 // This is a borrow even though it was not from our pool..
454 m_counter_borrows++;
455 // Guarenteed to count after the invocation to the callback
456 m_counter_ondemand_adds++;
457 // We're not performing any change to the original, just return it back.
458 return item;
459 });
460 }
461 else if (is_pool_starving()) {
462 // We're under-capacity.. but no dynamic resource provider
463 return std::unexpected(siddiqsoft::arrp::pool_error::UnderCapacityNoAutoGrow);
464 }
465 } // scope end
466 catch (std::exception& ex) {
467#if defined(DEBUG_TRACE)
468 std::print(std::cerr, "Error in borrow_from_pool: {}\n", ex.what());
469#endif
470 return std::unexpected(pool_error::Unknown);
471 }
472 catch (...) {
473 std::print(std::cerr, "UNKNOWN Error in borrow_from_pool\n");
474 return std::unexpected(pool_error::Unknown);
475 }
476
477 return std::unexpected(pool_error::NoMoreResources);
478 }

493 template <typename... Args>
494 auto seed_to_pool(Args&&... args) -> std::expected<void, pool_error>
495 {
496 std::scoped_lock l(m_pool_lock);
497
498 // Check inside the lock..
499 if (m_is_shutdown) return std::unexpected(pool_error::ShutdownInitiated);
500
501 m_pool.emplace_back(T {std::forward<Args&&>(args)...});
502 m_counter_seeds++;
503
504 // Update peak pool size for statistics
505 auto current_size = m_pool.size();
506 if (current_size > m_peak_poolsize.load()) {
507 m_peak_poolsize = current_size;
508 }
509
510 return {};
511 }

560 void return_to_pool(T&& item, bool isvalid)
561 {
562 std::scoped_lock l(m_pool_lock);
563
564 // Check inside the lock..
565 if (m_is_shutdown) return;
566
567 m_resources_checkedout--;
568
569 if (isvalid) {
570 m_pool.push_back(std::move(item));
571 m_counter_returns++;
572
573 // Update peak pool size for statistics
574 auto current_size = m_pool.size();
575 if (current_size > m_peak_poolsize.load()) {
576 m_peak_poolsize = current_size;
577 }
578 }
579 else {
580 m_counter_abandons++;
581 }
