154 void set_capacity(uint8_t init_capacity)
155 {
156#if defined(DEBUG)
157 std::print(std::cerr, "{} - capacity: {} init_capacity:{}\n", __func__, m_capacity, init_capacity);
158#endif
159
160 // We're going to be inside construction context and we're assured
161 // of only one invocation!
162 if (m_capacity == 0) {
163 if (init_capacity > resource_pool_limits::MaxCapacity) {
164 m_capacity = resource_pool_limits::MaxCapacity;
165 }
166 else if (init_capacity < resource_pool_limits::MinimumCapacity) {
167 m_capacity = resource_pool_limits::MinimumCapacity;
168 }
169 else {
170 m_capacity = init_capacity;
171 }
172#if defined(DEBUG)
173 std::print(std::cerr, "{} - capacity: {} init_capacity:{}\n", __func__, m_capacity, init_capacity);
174#endif
175 }

181 inline bool is_pool_starving() const { return m_resources_checkedout.load() + m_pool.size() < m_capacity; }

192 inline int64_t deficit_size() const

201 inline auto loan_size() const
202 {
203 auto loans = m_counter_borrows.load(); // total number of borrows (current counter)
204 loans -= m_counter_returns.load(); // total number of returns (current counter)
205 loans -= m_counter_abandons.load(); // adjust for any abandons
206 return loans;
207 }

283 template <typename... Args>
284 auto create_resource(Args&&... args) -> SRT
285 {
286 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
287 return SRT {[this](T&& src, bool isvalid) {
288 // this callback puts the resource back..
289 return this->return_to_pool(std::forward<T>(src), isvalid);
290 },
291 std::forward<Args>(args)...};
292 }

305 auto clear() -> pool_error
306 {
307 std::scoped_lock l(m_pool_lock);
308
309 try {
310 if (m_callback_on_resource_cleanup && !m_pool.empty()) {
311 while (!m_pool.empty()) {
312 RunOnEnd roe([&] {
313 m_pool.pop_front();
314 m_pool_semaphore.acquire();
315 });
316 // delegate to the cleanup.
317 // the delegate must not invoke any pool member to avoid deadlocks.
318 m_callback_on_resource_cleanup(std::move(m_pool.front()));
319 }
320 }
321 }
322 catch (std::exception& ex) {
323 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
324 }
325
326 // This will clear the pool (of any remaining resources) and reset the size to zero.
327 m_pool.clear();
328
329 return pool_error::Ok;
330 }

370 [[nodiscard]] auto borrow_from_pool(std::chrono::nanoseconds timeout = {}) -> SRT
371 {
372 try {
373 // @note We use a unique_lock vs a scoped_lock to allow ourselves
374 // to create the resource outside the lock!
375
376 // Now that we're inside the lock, we should check again to ensure that
377 // the shutdown is not in progress..
378 if (m_is_shutdown) return SRT {pool_error::ShutdownInitiated};
379
380 // If the pool is empty, we can check if we're under capacity and if so, we can create a new resource on-demand.
381 if (timeout.count() == 0 ? m_pool_semaphore.try_acquire() : m_pool_semaphore.try_acquire_for(timeout)) {
382 // We likely have a resource available.. grab a lock.
383 std::unique_lock l(m_pool_lock);
384 // We have a resource available.. just to be sure, we'll check the pool size inside the lock..
385 if (!m_pool.empty()) {
386 // Move the resource out of the pool while holding the lock.
387 T resource = std::move(m_pool.front());
388 m_pool.pop_front();
389
390 // Release the lock before constructing the scoped wrapper
391 // and updating the borrow counters.
392 l.unlock();
393
394 auto borrowed = create_resource(std::move(resource));
395 m_resources_checkedout++;
396 m_counter_borrows++;
397 return borrowed;
398 }
399 else {
400 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
401 }
402 }
403 else if (timeout.count() > 0) {
404 return SRT {siddiqsoft::arrp::pool_error::Timeout};
405 }
406 else {
407 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
408 }
409 } // scope end
410 catch (std::exception& ex) {
411 return SRT {siddiqsoft::arrp::pool_error::Unknown};
412 }
413 catch (...) {
414 std::print(std::cerr, "UNKNOWN Error in borrow_from_pool\n");
415 return SRT {siddiqsoft::arrp::pool_error::Unknown};
416 }
417
418 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
419 }

432 template <typename... Args>
433 auto seed_to_pool(Args&&... args) -> pool_error
434 {
435 {
436 std::scoped_lock l(m_pool_lock);
437
438 // Check inside the lock..
439 if (m_is_shutdown) return pool_error::ShutdownInitiated;
440
441 m_pool.emplace_back(std::forward<Args>(args)...);
442 m_counter_seeds++;
443
444 // Update peak pool size for statistics
445 auto current_size = m_pool.size();
446 if (current_size > m_peak_poolsize.load()) {
447 m_peak_poolsize = current_size;
448 }
449 }
450
451 // Signal the semaphore outside the lock to avoid potential deadlocks.
452 m_pool_semaphore.release(); // resource is available, increment semaphore
453
454 return pool_error::Ok;
455 }

509 void return_to_pool(T&& item, bool isvalid)
510 {
511 {
512 std::unique_lock l(m_pool_lock);
513
514 // Check inside the lock..
515 if (m_is_shutdown) return;
516
517 m_resources_checkedout--;
518
519 if (isvalid) {
520 m_pool.push_back(std::move(item));
521 m_counter_returns++;
522
523 // Update peak pool size for statistics
524 auto current_size = m_pool.size();
525 if (current_size > m_peak_poolsize.load()) {
526 m_peak_poolsize = current_size;
527 }
528
529 l.unlock();
530 // Signal the semaphore outside the lock to avoid potential deadlocks.
531 if (isvalid) m_pool_semaphore.release(); // resource is available, increment semaphore
532 }
533 else {
534 m_counter_abandons++;
535 }
536 }
