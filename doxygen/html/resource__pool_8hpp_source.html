295 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
296 return resource_guard<T> {[this](T&& src, bool isvalid) {

314 template <typename F, typename... Args>
315 auto create_from_callback(F&& f, Args&&... args) -> resource_guard<T>
316 {
317 using result_t = std::invoke_result_t<F, Args...>;
318
319 if constexpr (std::is_same_v<result_t, resource_guard<T>>) {
320 return std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
321 }
322 else if constexpr (std::is_same_v<result_t, T>) {
323 return make_resource_guard(std::invoke(std::forward<F>(f), std::forward<Args>(args)...));
324 }
325 else {
326 static_assert(std::is_same_v<result_t, resource_guard<T>> || std::is_same_v<result_t, T>,
327 "Callback must return either resource_guard<T> or T");

341 template <typename F>
342 void set_factory_callback(F&& f)
343 {
344 auto factory = std::forward<F>(f);
345 auto wrapped = [this, factory = std::move(factory)]() mutable -> resource_guard<T> {
346 return create_from_callback(factory);
347 };
348
349 // Synchronize with borrow_impl()'s reads of m_factory_callback (both the
350 // locked and unlocked read sites) so concurrent set_factory_callback()
351 // calls do not race with in-flight borrows.
352 std::scoped_lock l(m_pool_lock);
353 m_factory_callback = std::move(wrapped);
354 }

371 auto clear() -> pool_error
372 {
373 std::scoped_lock l(m_pool_lock);
374
375 // Drain semaphore and pool together so their counts stay in sync.
376 // The cleanup callback, if set, runs under the lock per item.
377 //
378 // @note We MUST use try_acquire() here, not the blocking acquire().
379 // A concurrent borrow_impl() may have already claimed a permit via
380 // m_pool_semaphore.try_acquire()/try_acquire_for() and be waiting on
381 // m_pool_lock (which we currently hold) before it pops its item from
382 // m_pool. If we block here waiting for a permit that will only be
383 // released once that borrower runs — and it can't run until we
384 // release this lock — the pool deadlocks. When try_acquire() fails,
385 // every remaining permit is already claimed by such an in-flight
386 // borrower, so we stop draining and let those borrowers pop their
387 // own items once they acquire the lock after we release it.
388 while (!m_pool.empty() && m_pool_semaphore.try_acquire()) {
389 auto item = std::move(m_pool.front());
390 m_pool.pop_front();
391 if (m_callback_on_resource_cleanup) {
392 try {
393 m_callback_on_resource_cleanup(item);
394 }
395 catch (std::exception& ex) {
396 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
397 }
398 catch (...) {
399 std::print(std::cerr, "{} - unknown exception while delegating to on_cleanup\n", __func__);
400 }
401 }
402 }
403
404 return pool_error::Ok;
405 }

440 [[nodiscard]] auto borrow_impl(std::chrono::nanoseconds timeout = {}, bool createIfEmptyTimeout = false)
441 -> resource_guard<T>
442 {
443 try {
444 // @note We use a unique_lock vs a scoped_lock to allow ourselves
445 // to create the resource outside the lock!
446
447 // Fast pre-check — the authoritative shutdown check is performed
448 // under the lock below after semaphore acquisition.
449 if (m_is_shutdown) return resource_guard<T> {pool_error::ShutdownInitiated};
450
451 // Wait for an available resource, then optionally create one when
452 // try_borrow_create() was requested.
453 if (timeout.count() == 0 ? m_pool_semaphore.try_acquire() : m_pool_semaphore.try_acquire_for(timeout)) {
454 // We likely have a resource available.. grab a lock.
455 std::unique_lock l(m_pool_lock);
456
457 // Authoritative shutdown check under the lock: the destructor may
458 // have run between the pre-semaphore check and taking this lock.
459 if (m_is_shutdown) {
460 m_pool_semaphore.release();
461 return resource_guard<T> {pool_error::ShutdownInitiated};
462 }
463
464 // We have a resource available.. just to be sure, we'll check the pool size inside the lock..
465 if (!m_pool.empty()) {
466 // Move the resource out of the pool while holding the lock.
467 auto borrowed = make_resource_guard(std::move(m_pool.front()));
468 // Remove from the deque.. only one client may have exclusive use..
469 m_pool.pop_front();
470
471 // Increment checkout and borrow counters under the lock to prevent
472 // state accounting underflow / race conditions with concurrent return_to_pool()
473 m_resources_checkedout++;
474 m_counter_borrows++;
475
476 l.unlock();
477
478 return borrowed;
479 }
480 else if (createIfEmptyTimeout && m_factory_callback) {
481 std::println(std::cerr, "{} - Empty pool; asked to create new if empty..", __func__);
482 // Release the lock before invoking arbitrary user code to
483 // prevent deadlock if the factory calls back into the pool.
484 auto cb = m_factory_callback;
485 l.unlock();
486 auto ondemand = create_from_callback(cb);
487 m_counter_ondemand_adds++;
488 m_resources_checkedout++;
489 m_counter_borrows++;
490 return ondemand;
491 }
492 else {
493 // Permit was acquired but no resource exists; release it so
494 // the semaphore count stays in sync with the pool size.
495 m_pool_semaphore.release();
496 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
497 }
498 }
499 else if (createIfEmptyTimeout) {
500 // Take a copy of the factory under the lock so the read is
501 // synchronized with set_factory_callback(), then invoke it
502 // without holding m_pool_lock to avoid deadlock if the factory
503 // calls back into the pool.
504 std::function<resource_guard<T>()> cb;
505 {
506 std::scoped_lock l(m_pool_lock);
507 cb = m_factory_callback;
508 }
509 if (!cb) {
510 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
511 }
512 std::println(std::cerr, "{} - We exhausted timeout; asked to create new if empty..", __func__);
513 auto ondemand = create_from_callback(cb);
514 m_counter_ondemand_adds++;
515 m_resources_checkedout++;
516 m_counter_borrows++;
517 return ondemand;
518 }
519 else if (timeout.count() > 0) {
520 return resource_guard<T> {siddiqsoft::arrp::pool_error::Timeout};
521 }
522 else {
523 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
524 }
525 } // scope end
526 catch (std::exception& ex) {
527 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
528 }
529 catch (...) {
530 std::println(std::cerr, "UNKNOWN Error in borrow");
531 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
532 }
533
534 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
535 }

562 template <typename... Args>
563 auto seed(Args&&... args) -> pool_error
564 {
565 {
566 std::scoped_lock l(m_pool_lock);
567
568 // Check inside the lock..
569 if (m_is_shutdown) return pool_error::ShutdownInitiated;
570
571 m_pool.emplace_back(std::forward<Args>(args)...);
572 m_counter_seeds++;
573
574 // Update peak pool size for statistics
575 auto current_size = m_pool.size();
576 if (current_size > m_peak_poolsize.load()) {
577 m_peak_poolsize = current_size;
578 }
579
580 // Signal semaphore while holding the lock to maintain atomic visibility
581 m_pool_semaphore.release();
582 }
583
584 return pool_error::Ok;
585 }

631 void return_to_pool(T&& item, bool isvalid)
632 {
633 std::unique_lock l(m_pool_lock);
634
635 // Always decrement the checkout counter — the resource is leaving our
636 // custody regardless of whether the pool is still alive.
637 m_resources_checkedout--;
638
639 // Check inside the lock..
640 if (m_is_shutdown) return;
641
642 if (isvalid) {
643 m_pool.push_back(std::move(item));
644 m_counter_returns++;
645
646 // Update peak pool size for statistics
647 auto current_size = m_pool.size();
648 if (current_size > m_peak_poolsize.load()) {
649 m_peak_poolsize = current_size;
650 }
651
652 // Signal semaphore while holding the lock to maintain atomic visibility
653 m_pool_semaphore.release();
654 }
655 else {
656 m_counter_abandons++;
657 }
658 }
