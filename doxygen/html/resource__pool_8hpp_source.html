356 [[nodiscard]] auto borrow_from_pool() -> SRT
357 {
358 try {
359 // @note We use a unique_lock vs a scoped_lock to allow ourselves
360 // to create the resource outside the lock!
361 std::unique_lock l(m_pool_lock);
362
363 if (!m_pool.empty()) {
364 // The pool is non-empty; return from the pool
365 // Return first element from the pool and pop it on scope end
366 RunOnEnd pop_guard([&]() {
367 m_pool.pop_front();
368 });
369
370 m_resources_checkedout++;
371 ++m_counter_borrow_from_pool;
372
373
374 // Make a wrapper..
375 // Create a SRT element and wire up the auto-return callback to return
376 // the resource back to this object.
377 return SRT {std::move(m_pool.front()), [this](T&& src) {
378 this->m_counter_auto_returned++;
379 this->return_to_pool(std::forward<T&&>(src));
380 }};
381 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
382 // return temp;
383
384 // The pop_front() happens within this scope and within the lock!
385 }
386 else if ((m_capacity > m_pool.size() + m_resources_checkedout) && m_callback_to_add_new_raw_resource_to_pool) {
387 // We have no more items in the pool (we're starting up or everything is
388 // checked out) but we have not reached the limit. The limit is number
389 // of m_resources_checkedout + pool.size() < m_capacity We are
390 // under-capacity.. so we can return to the caller a new item..
391 m_resources_checkedout++;
392 // We should unlock the resource and ..
393 l.unlock();
394
395 ++m_counter_borrow_from_pool;
396 // Update the attempted delegated calls to add new raw resource to pool.
397 ++m_counter_ondemand_adds;
398
399 // ..delegate the new resource acquisition
400 // outside the lock.
401 return m_callback_to_add_new_raw_resource_to_pool(*this);
402 }
403 else if (m_capacity > m_pool.size() + m_resources_checkedout) {
404 // We're under-capacity.. but no dynamic resource provider
405 }
406 } // scope end
407 catch (const std::exception& ex) {
408 std::cerr << std::format("Error in borrow_from_pool: {}\n", ex.what());
409 throw;
410 }
411 catch (...) {
412 std::cerr << "Unknown exception in borrow_from_pool\n";
413 throw;
414 }
415
416 auto msg = std::format(
417 "Pool Size:{} checkedout:{} capacity:{}", m_pool.size(), m_resources_checkedout.load(), m_capacity);
418 throw std::runtime_error(msg);
419 }
