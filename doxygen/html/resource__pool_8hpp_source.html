271 template <typename... Args>
272 auto make_resource_guard(Args&&... args) -> resource_guard<T>
273 {
274 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
275 return resource_guard<T> {[this](T&& src, bool isvalid) {
276 // this callback puts the resource back..
277 return this->return_to_pool(std::forward<T>(src), isvalid);

293 template <typename F, typename... Args>
294 auto create_from_callback(F&& f, Args&&... args) -> resource_guard<T>
295 {
296 using result_t = std::invoke_result_t<F, Args...>;
297
298 if constexpr (std::is_same_v<result_t, resource_guard<T>>) {
299 return std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
300 }
301 else if constexpr (std::is_same_v<result_t, T>) {
302 return make_resource_guard(std::invoke(std::forward<F>(f), std::forward<Args>(args)...));
303 }
304 else {
305 static_assert(std::is_same_v<result_t, resource_guard<T>> || std::is_same_v<result_t, T>,
306 "Callback must return either resource_guard<T> or T");

320 template <typename F>
321 void set_factory_callback(F&& f)
322 {
323 auto factory = std::forward<F>(f);
324 auto wrapped = [this, factory = std::move(factory)]() mutable -> resource_guard<T> {
325 return create_from_callback(factory);
326 };
327
328 // Synchronize with borrow_impl()'s reads of m_factory_callback (both the
329 // locked and unlocked read sites) so concurrent set_factory_callback()
330 // calls do not race with in-flight borrows.
331 std::scoped_lock l(m_pool_lock);
332 m_factory_callback = std::move(wrapped);
333 }

350 auto clear() -> pool_error
351 {
352 std::scoped_lock l(m_pool_lock);
353
354 try {
355 // Drain semaphore and pool together so their counts stay in sync.
356 // The cleanup callback, if set, runs under the lock per item.
357 //
358 // @note We MUST use try_acquire() here, not the blocking acquire().
359 // A concurrent borrow_impl() may have already claimed a permit via
360 // m_pool_semaphore.try_acquire()/try_acquire_for() and be waiting on
361 // m_pool_lock (which we currently hold) before it pops its item from
362 // m_pool. If we block here waiting for a permit that will only be
363 // released once that borrower runs — and it can't run until we
364 // release this lock — the pool deadlocks. When try_acquire() fails,
365 // every remaining permit is already claimed by such an in-flight
366 // borrower, so we stop draining and let those borrowers pop their
367 // own items once they acquire the lock after we release it.
368 while (!m_pool.empty() && m_pool_semaphore.try_acquire()) {
369 auto item = std::move(m_pool.front());
370 m_pool.pop_front();
371 if (m_callback_on_resource_cleanup) {
372 // delegate to the cleanup.
373 // the delegate must not invoke any pool member to avoid deadlocks.
374 m_callback_on_resource_cleanup(item);
375 }
376 }
377 }
378 catch (std::exception& ex) {
379 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
380 }
381
382 return pool_error::Ok;
383 }

418 [[nodiscard]] auto borrow_impl(std::chrono::nanoseconds timeout = {}, bool createIfEmptyTimeout = false)
419 -> resource_guard<T>
420 {
421 try {
422 // @note We use a unique_lock vs a scoped_lock to allow ourselves
423 // to create the resource outside the lock!
424
425 // Fast pre-check — the authoritative shutdown check is performed
426 // under the lock below after semaphore acquisition.
427 if (m_is_shutdown) return resource_guard<T> {pool_error::ShutdownInitiated};
428
429 // Wait for an available resource, then optionally create one when
430 // try_borrow_create() was requested.
431 if (timeout.count() == 0 ? m_pool_semaphore.try_acquire() : m_pool_semaphore.try_acquire_for(timeout)) {
432 // We likely have a resource available.. grab a lock.
433 std::unique_lock l(m_pool_lock);
434
435 // Authoritative shutdown check under the lock: the destructor may
436 // have run between the pre-semaphore check and taking this lock.
437 if (m_is_shutdown) {
438 m_pool_semaphore.release();
439 return resource_guard<T> {pool_error::ShutdownInitiated};
440 }
441
442 // We have a resource available.. just to be sure, we'll check the pool size inside the lock..
443 if (!m_pool.empty()) {
444 // Move the resource out of the pool while holding the lock.
445 auto borrowed = make_resource_guard(std::move(m_pool.front()));
446 // Remove from the deque.. only one client may have exclusive use..
447 m_pool.pop_front();
448
449 // Release the lock before constructing the scoped wrapper
450 // and updating the borrow counters.
451 l.unlock();
452
453 m_resources_checkedout++;
454 m_counter_borrows++;
455 return borrowed;
456 }
457 else if (createIfEmptyTimeout && m_factory_callback) {
458 std::println(std::cerr, "{} - Empty pool; asked to create new if empty..", __func__);
459 // Release the lock before invoking arbitrary user code to
460 // prevent deadlock if the factory calls back into the pool.
461 auto cb = m_factory_callback;
462 l.unlock();
463 auto ondemand = create_from_callback(cb);
464 m_counter_ondemand_adds++;
465 m_resources_checkedout++;
466 m_counter_borrows++;
467 return ondemand;
468 }
469 else {
470 // Permit was acquired but no resource exists; release it so
471 // the semaphore count stays in sync with the pool size.
472 m_pool_semaphore.release();
473 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
474 }
475 }
476 else if (createIfEmptyTimeout) {
477 // Take a copy of the factory under the lock so the read is
478 // synchronized with set_factory_callback(), then invoke it
479 // without holding m_pool_lock to avoid deadlock if the factory
480 // calls back into the pool.
481 std::function<resource_guard<T>()> cb;
482 {
483 std::scoped_lock l(m_pool_lock);
484 cb = m_factory_callback;
485 }
486 if (!cb) {
487 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
488 }
489 std::println(std::cerr, "{} - We exhausted timeout; asked to create new if empty..", __func__);
490 auto ondemand = create_from_callback(cb);
491 m_counter_ondemand_adds++;
492 m_resources_checkedout++;
493 m_counter_borrows++;
494 return ondemand;
495 }
496 else if (timeout.count() > 0) {
497 return resource_guard<T> {siddiqsoft::arrp::pool_error::Timeout};
498 }
499 else {
500 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
501 }
502 } // scope end
503 catch (std::exception& ex) {
504 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
505 }
506 catch (...) {
507 std::println(std::cerr, "UNKNOWN Error in borrow");
508 return resource_guard<T> {siddiqsoft::arrp::pool_error::Unknown};
509 }
510
511 return resource_guard<T> {siddiqsoft::arrp::pool_error::NoMoreResources};
512 }

539 template <typename... Args>
540 auto seed(Args&&... args) -> pool_error
541 {
542 {
543 std::scoped_lock l(m_pool_lock);
544
545 // Check inside the lock..
546 if (m_is_shutdown) return pool_error::ShutdownInitiated;
547
548 m_pool.emplace_back(std::forward<Args>(args)...);
549 m_counter_seeds++;
550
551 // Update peak pool size for statistics
552 auto current_size = m_pool.size();
553 if (current_size > m_peak_poolsize.load()) {
554 m_peak_poolsize = current_size;
555 }
556 }
557
558 // Signal the semaphore outside the lock to avoid potential deadlocks.
559 m_pool_semaphore.release(); // resource is available, increment semaphore
560
561 return pool_error::Ok;
562 }

597 protected:
608 void return_to_pool(T&& item, bool isvalid)
609 {
610 std::unique_lock l(m_pool_lock);
611
612 // Always decrement the checkout counter — the resource is leaving our
613 // custody regardless of whether the pool is still alive.
614 m_resources_checkedout--;
615
616 // Check inside the lock..
617 if (m_is_shutdown) return;
618
619 if (isvalid) {
620 m_pool.push_back(std::move(item));
621 m_counter_returns++;
622
623 // Update peak pool size for statistics
624 auto current_size = m_pool.size();
625 if (current_size > m_peak_poolsize.load()) {
626 m_peak_poolsize = current_size;
627 }
628
629 l.unlock();
630 // Signal the semaphore outside the lock to avoid potential deadlocks.
631 if (isvalid) m_pool_semaphore.release(); // resource is available, increment semaphore
632 }
633 else {
634 m_counter_abandons++;
635 }
636 }
