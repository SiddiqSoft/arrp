202 inline auto loan_size() const
203 {
204 auto loans = m_counter_borrows.load(); // total number of borrows (current counter)
205 loans -= m_counter_returns.load(); // total number of returns (current counter)
206 loans -= m_counter_abandons.load(); // adjust for any abandons

284 template <typename... Args>
285 auto wrap_as_scoped_resource(Args&&... args) -> SRT
286 {
287 // Allow the compiler to use NRVO (move elision; do not use std::move here!)
288 return SRT {[this](T&& src, bool isvalid) {
289 // this callback puts the resource back..
290 return this->return_to_pool(std::forward<T>(src), isvalid);
291 },
292 std::forward<Args>(args)...};
293 }

295 // Helper: type alias for a user-provided callback that returns `SRT` and accepts `Args...`.
296 template <typename... Args>
297 using resource_callback_t = std::function<SRT(Args...)>;
298
299 // Helper to create a resource from a user-provided callback.
300 // The callback may return either `SRT` (already-wrapped) or `T` (raw resource).
301 // If it returns `T`, we forward the result to `wrap_as_scoped_resource` to obtain an `SRT`.
302 template <typename F, typename... Args>
303 auto create_from_callback(F&& f, Args&&... args) -> SRT
304 {
305 using result_t = std::invoke_result_t<F, Args...>;
306
307 if constexpr (std::is_same_v<result_t, SRT>) {
308 return std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
309 }
310 else if constexpr (std::is_same_v<result_t, T>) {
311 return wrap_as_scoped_resource(std::invoke(std::forward<F>(f), std::forward<Args>(args)...));
312 }
313 else {
314 static_assert(std::is_same_v<result_t, SRT> || std::is_same_v<result_t, T>, "Callback must return either SRT or T");
315 }
316 }

339 auto clear() -> pool_error
340 {
341 std::scoped_lock l(m_pool_lock);
342
343 try {
344 if (m_callback_on_resource_cleanup && !m_pool.empty()) {
345 while (!m_pool.empty()) {
346 m_pool_semaphore.acquire();
347 auto item = std::move(m_pool.front());
348 m_pool.pop_front();
349 // delegate to the cleanup.
350 // the delegate must not invoke any pool member to avoid deadlocks.
351 m_callback_on_resource_cleanup(item);
352 }
353 }
354 }
355 catch (std::exception& ex) {
356 std::print(std::cerr, "{} - exception while delegating to on_cleanup: {}\n", __func__, ex.what());
357 }
358
359 // This will clear the pool (of any remaining resources) and reset the size to zero.
360 m_pool.clear();
361
362 return pool_error::Ok;
363 }

403 [[nodiscard]] auto borrow_from_pool(std::chrono::nanoseconds timeout = {}, bool createIfEmptyTimeout = false) -> SRT
404 {
405 try {
406 // @note We use a unique_lock vs a scoped_lock to allow ourselves
407 // to create the resource outside the lock!
408
409 // Now that we're inside the lock, we should check again to ensure that
410 // the shutdown is not in progress..
411 if (m_is_shutdown) return SRT {pool_error::ShutdownInitiated};
412
413 // If the pool is empty, we can check if we're under capacity and if so, we can create a new resource on-demand.
414 if (timeout.count() == 0 ? m_pool_semaphore.try_acquire() : m_pool_semaphore.try_acquire_for(timeout)) {
415 // We likely have a resource available.. grab a lock.
416 std::unique_lock l(m_pool_lock);
417 // We have a resource available.. just to be sure, we'll check the pool size inside the lock..
418 if (!m_pool.empty()) {
419 // Move the resource out of the pool while holding the lock.
420 auto borrowed = wrap_as_scoped_resource(std::move(m_pool.front()));
421 // Remove from the deque.. only one client may have exclusive use..
422 m_pool.pop_front();
423
424 // Release the lock before constructing the scoped wrapper
425 // and updating the borrow counters.
426 l.unlock();
427
428 m_resources_checkedout++;
429 m_counter_borrows++;
430 return borrowed;
431 }
432 else if (createIfEmptyTimeout && m_factory_callback) {
433 std::println(std::cerr, "{} - Empty pool; asked to create new if empty..", __func__);
434 auto ondemand = create_from_callback(m_factory_callback);
435 m_counter_ondemand_adds++;
436 m_resources_checkedout++;
437 m_counter_borrows++;
438 return ondemand;
439 }
440 else {
441 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
442 }
443 }
444 else if (createIfEmptyTimeout && m_factory_callback) {
445 std::println(std::cerr, "{} - We exhausted timeout; asked to create new if empty..", __func__);
446 auto ondemand = create_from_callback(m_factory_callback);
447 m_counter_ondemand_adds++;
448 m_resources_checkedout++;
449 m_counter_borrows++;
450 return ondemand;
451 }
452 else if (timeout.count() > 0) {
453 return SRT {siddiqsoft::arrp::pool_error::Timeout};
454 }
455 else {
456 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
457 }
458 } // scope end
459 catch (std::exception& ex) {
460 return SRT {siddiqsoft::arrp::pool_error::Unknown};
461 }
462 catch (...) {
463 std::println(std::cerr, "UNKNOWN Error in borrow_from_pool");
464 return SRT {siddiqsoft::arrp::pool_error::Unknown};
465 }
466
467 return SRT {siddiqsoft::arrp::pool_error::NoMoreResources};
468 }

481 template <typename... Args>
482 auto seed_to_pool(Args&&... args) -> pool_error
483 {
484 {
485 std::scoped_lock l(m_pool_lock);
486
487 // Check inside the lock..
488 if (m_is_shutdown) return pool_error::ShutdownInitiated;
489
490 m_pool.emplace_back(std::forward<Args>(args)...);
491 m_counter_seeds++;
492
493 // Update peak pool size for statistics
494 auto current_size = m_pool.size();
495 if (current_size > m_peak_poolsize.load()) {
496 m_peak_poolsize = current_size;
497 }
498 }
499
500 // Signal the semaphore outside the lock to avoid potential deadlocks.
501 m_pool_semaphore.release(); // resource is available, increment semaphore
502
503 return pool_error::Ok;
504 }

558 void return_to_pool(T&& item, bool isvalid)
559 {
560 std::unique_lock l(m_pool_lock);
561
562 // Check inside the lock..
563 if (m_is_shutdown) return;
564
565 m_resources_checkedout--;
566
567 if (isvalid) {
568 m_pool.push_back(std::move(item));
569 m_counter_returns++;
570
571 // Update peak pool size for statistics
572 auto current_size = m_pool.size();
573 if (current_size > m_peak_poolsize.load()) {
574 m_peak_poolsize = current_size;
575 }
576
577 l.unlock();
578 // Signal the semaphore outside the lock to avoid potential deadlocks.
579 if (isvalid) m_pool_semaphore.release(); // resource is available, increment semaphore
580 }
581 else {
582 m_counter_abandons++;
583 }
584 }
